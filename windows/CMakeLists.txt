cmake_minimum_required(VERSION 3.20)
project(SSMVCore LANGUAGES CXX)
add_library(ssmv_core Core/Markdown.cpp Core/Documents.cpp Core/Session.cpp)
target_include_directories(ssmv_core PUBLIC Core)
target_compile_features(ssmv_core PUBLIC cxx_std_20)
if(MSVC)
    target_compile_options(ssmv_core PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(ssmv_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(ssmv_core_tests Tests/CoreTests.cpp)
    target_link_libraries(ssmv_core_tests PRIVATE ssmv_core)
    if(MSVC)
        target_compile_options(ssmv_core_tests PRIVATE /utf-8)
    endif()
    add_test(NAME ssmv_core_tests COMMAND ssmv_core_tests)
endif()
if(BUILD_TESTING)
    add_executable(ssmv_markdown_tests Tests/MarkdownTests.cpp)
    target_link_libraries(ssmv_markdown_tests PRIVATE ssmv_core)
    add_test(NAME ssmv_markdown_tests COMMAND ssmv_markdown_tests)
endif()

if(BUILD_TESTING)
    add_executable(ssmv_session_tests Tests/SessionTests.cpp)
    target_link_libraries(ssmv_session_tests PRIVATE ssmv_core)
    if(MSVC)
        target_compile_options(ssmv_session_tests PRIVATE /utf-8)
    endif()
    add_test(NAME ssmv_session_tests COMMAND ssmv_session_tests)
endif()
